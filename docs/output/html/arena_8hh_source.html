55 [[nodiscard]] void* allocate(
56 const std::size_t size,
57 const std::size_t align = alignof(std::max_align_t)
58 ) {
59 assert(std::has_single_bit(align) && "alignment must be n^2");
60
61 if (size == 0) [[unlikely]] return nullptr;
62
63 for (;;) {
64 if (!m_slabs.empty()) {
65 auto& s = m_slabs.back();
66 auto* p = std::bit_cast<std::byte*>(
67 (std::bit_cast<std::uintptr_t>(s.cursor) + align - 1) & ~(align - 1)
68 );
69
70 if (p + size <= s.end) {
71 s.cursor = p + size;
72 return p;
73 }
74 }
75
76 m_slabs.emplace_back(std::max(size + align, m_slab_size));
77 }
78 }
