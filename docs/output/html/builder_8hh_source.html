14 static std::span<T> copy_to_arena(support::Arena& arena, const std::vector<T>& src) {
15 if (src.empty()) return {};
16 auto dst = arena.alloc_array_uninit<T>(src.size());
17 std::ranges::copy(src, dst.begin());
18 return dst;
19 }

115 void finalise() {
116 auto& arena = *m_fn.arena;
117 for (std::size_t i = 0; i < m_blocks.size(); ++i) {
118 const auto start = m_block_inst_starts[i];
119 const auto end = (i + 1 < m_block_inst_starts.size())
120 ? m_block_inst_starts[i + 1]
121 : static_cast<uint32_t>(m_insts.size());
122 m_blocks[i].insts_offset = start;
123 m_blocks[i].insts_count = static_cast<uint16_t>(end - start);
124 }
125
126 m_fn.blocks = copy_to_arena<IRBlock>(arena, m_blocks);
127 m_fn.insts = copy_to_arena<IRInst>(arena, m_insts);
128 m_fn.operands = copy_to_arena<ValueRef>(arena, m_operands);
129 m_fn.succs = copy_to_arena<BlockRef>(arena, m_succs);
130 m_fn.values = copy_to_arena<Value>(arena, m_values);
131 m_fn.block_params = copy_to_arena<ValueRef>(arena, m_block_params);
132 }

153 [[nodiscard]] ValueRef emit(
154 const IROp op,
155 const Type ty,
156 const std::span<const ValueRef> ops,
157 const uint64_t imm = 0,
158 const ICmpCC cc = {}
159 ) {
160 const auto inst_id = static_cast<uint32_t>(m_insts.size());
161 const auto vref = push_value(ty, InstRef{ inst_id });
162
163 IRInst inst{};
164 inst.op = op;
165 inst.cc = cc;
166 inst.result = vref;
167 inst.ops_offset = static_cast<uint32_t>(m_operands.size());
168 inst.ops_count = static_cast<uint16_t>(ops.size());
169 inst.imm = imm;
170
171 for (const auto op_ref : ops) m_operands.push_back(op_ref);
172 m_insts.push_back(inst);
173 return vref;
174 }

181 void emit_terminator(
182 const IROp op,
183 const BlockRef primary,
184 const BlockRef secondary,
185 const ValueRef cond,
186 const std::span<const ValueRef> primary_args,
187 const std::span<const ValueRef> secondary_args
188 ) {
189 auto& blk = m_blocks[m_cur.id];
190 blk.succs_offset = static_cast<uint32_t>(m_succs.size());
191
192 std::vector<ValueRef> ops;
193 if (cond.id != ~0u) ops.push_back(cond);
194 for (const auto a : primary_args) ops.push_back(a);
195 for (const auto a : secondary_args) ops.push_back(a);
196
197 m_succs.push_back(primary);
198 if (op == IROp::brif) {
199 m_succs.push_back(secondary);
200 blk.succs_count = 2;
201 } else {
202 blk.succs_count = 1;
203 }
204
205 emit(op, Type::unit, ops);
206 }
207 };
