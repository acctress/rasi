56 out = std::format_to(out, "v{}", vref.id);

59
60 for (uint16_t ii = 0; ii < b.insts_count; ++ii) {
61 const auto& inst = fn.insts[b.insts_offset + ii];
62 out = std::format_to(out, " v{} = {}", inst.result.id, inst.op);
63
64 if (inst.op == rasi::ir::IROp::iconst) {
65 out = std::format_to(out, " #{}", inst.imm);
66 } else if (inst.op == rasi::ir::IROp::icmp) {
67 out = std::format_to(out, " {}", inst.cc);
68 }
69
70 for (uint16_t oi = 0; oi < inst.ops_count; ++oi) {
71 out = std::format_to(out, " v{}", fn.operands[inst.ops_offset + oi].id);
72 }
73 out = std::format_to(out, "\n");
74 }
75
76 for (uint8_t si = 0; si < b.succs_count; ++si)
