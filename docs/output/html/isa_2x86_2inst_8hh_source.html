286 [[nodiscard]] std::string to_string() const {
287 return std::visit([](const auto &i) -> std::string {
288 using T = std::decay_t<decltype(i)>;
289
290 if constexpr (std::is_same_v<T, inst::Alu>) {
291 return std::format("{} {}, {}", x86_64::to_string(i.op), i.src, i.dst);
292 } else if constexpr (std::is_same_v<T, inst::AluImm>) {
293 return std::format("{} ${}, {}", x86_64::to_string(i.op), i.imm, i.dst);
294 } else if constexpr (std::is_same_v<T, inst::Mov>) {
295 return std::format("mov {}, {}", i.src, i.dst);
296 } else if constexpr (std::is_same_v<T, inst::MovImm>) {
297 return std::format("mov ${}, {}", i.imm, i.dst);
298 } else if constexpr (std::is_same_v<T, inst::MovLoad>) {
299 return std::format("mov {}({}), {}", i.mem.disp, i.mem.base, i.dst);
300 } else if constexpr (std::is_same_v<T, inst::MovStore>) {
301 return std::format("mov {}, {}({})", i.src, i.mem.disp, i.mem.base);
302 } else if constexpr (std::is_same_v<T, inst::Movzx>) {
303 return std::format("movz{}q {}, {}", x86_64::to_string(i.size), i.src, i.dst);
304 } else if constexpr (std::is_same_v<T, inst::Movsx>) {
305 return std::format("movs{}q {}, {}", x86_64::to_string(i.size), i.src, i.dst);
306 } else if constexpr (std::is_same_v<T, inst::Cdq>) {
307 return "cdq";
308 } else if constexpr (std::is_same_v<T, inst::Div>) {
309 return std::format("div {}", i.src);
310 } else if constexpr (std::is_same_v<T, inst::Idiv>) {
311 return std::format("idiv {}", i.src);
312 } else if constexpr (std::is_same_v<T, inst::Push>) {
313 return std::format("push {}", i.src);
314 } else if constexpr (std::is_same_v<T, inst::Pop>) {
315 return std::format("pop {}", i.dst);
316 } else if constexpr (std::is_same_v<T, inst::LeaFrame>) {
317 return std::format("lea {}(%rbp), {}", i.offset, i.dst);
318 } else if constexpr (std::is_same_v<T, inst::SetCC>) {
319 return std::format("set{} {}", x86_64::to_string(i.cc), i.dst);
320 } else if constexpr (std::is_same_v<T, inst::Jmp>) {
321 return std::format("jmp .L{}", i.target.id);
322 } else if constexpr (std::is_same_v<T, inst::Jcc>) {
323 return std::format("j{} .L{}", x86_64::to_string(i.cc), i.target.id);
324 } else if constexpr (std::is_same_v<T, inst::Call>) {
325 return std::format("call .Lfunc{}", i.func_id);
326 } else if constexpr (std::is_same_v<T, inst::Ret>) {
327 return "ret";
328 } else if constexpr (std::is_same_v<T, inst::DummyUse>) {
329 return std::format("# keep-alive {}", i.reg);
330 }
331 }, inner);
332 }
