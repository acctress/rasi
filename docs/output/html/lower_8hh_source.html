29 [[nodiscard]] VCode<X86Inst> lower(
30 ir::IRFunction &fn,
31 CallConv cc = CallConv::SystemV
32 ) {
33 m_fn = &fn;
34 m_vreg_c = 0;
35 m_value_regs.clear();
36 m_builder = codegen::VCodeBuilder<X86Inst>{};
37
38 std::vector<ir::Type> param_tys;
39 std::vector<ir::Type> ret_tys;
40
41 param_tys.reserve(fn.sig.params_count);
42 for (uint16_t i = 0; i < fn.sig.params_count; ++i)
43 param_tys.push_back(fn.values[i].type);
44
45 if (fn.sig.ret != ir::Type::unit)
46 ret_tys.push_back(fn.sig.ret);
47
48 m_sig = m_abi.compute(cc, param_tys, ret_tys);
49
50 m_value_regs.resize(fn.values.size());
51 for (auto i{0u}; i < fn.values.size(); ++i)
52 m_value_regs[i] = alloc_vreg(fn.values[i].type);
53
54 visit_function(fn);
55
56 m_fn = nullptr;
57 return std::move(m_builder).finalise();
58 }

74 void on_inst(ir::IRFunction &, ir::IRBlock &, ir::IRInst &inst) {
75 using enum ir::IROp;
76 switch (inst.op) {
77 case iconst: lower_iconst(inst);
78 break;
79 case iadd: lower_binop(inst, AluOp::Add);
80 break;
81 case isub: lower_binop(inst, AluOp::Sub);
82 break;
83 case imul: lower_binop(inst, AluOp::Mul);
84 break;
85 case ret: lower_ret(inst);
86 break;
87 case idiv:
88 case icmp:
89 case br:
90 case brif:
91 case call: std::unreachable();
92 }
93 }

120 void lower_iconst(const ir::IRInst &inst) {
121 m_builder.push(inst::MovImm{
122 .dst = reg_for(inst.result),
123 .imm = static_cast<int64_t>(inst.imm),
124 });
125 }
126
127 void lower_binop(const ir::IRInst &inst, AluOp op) {
128 const auto dst = reg_for(inst.result);
129 const auto lhs = reg_for(m_fn->operands[inst.ops_offset]);
130 const auto rhs = reg_for(m_fn->operands[inst.ops_offset + 1]);
131
132 m_builder.push(inst::Mov{.dst = dst, .src = lhs});
133 m_builder.push(inst::Alu{.op = op, .dst = dst, .src = rhs});
134 }
135
136 void lower_ret(const ir::IRInst &inst) {
137 if (inst.ops_count > 0) {
138 assert(!m_sig.rets.empty() && m_sig.rets[0].is_reg());
139
140 const auto src = reg_for(m_fn->operands[inst.ops_offset]);
141 const auto ret_preg = m_sig.rets[0].reg();
142
143 m_builder.push(inst::Mov{
144 .dst = codegen::Reg::from_preg(ret_preg),
145 .src = src,
146 });
147 }
148
149 m_builder.push(inst::Ret{});
150 }
