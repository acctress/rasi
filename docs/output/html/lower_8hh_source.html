28 [[nodiscard]] codegen::VCode<X86Inst> lower(ir::IRFunction& fn) {
29 m_fn = &fn;
30 m_vreg_c = 0;
31 m_value_regs.clear();
32 m_builder = codegen::VCodeBuilder<X86Inst>{};
33
34 m_value_regs.resize(fn.values.size());
35 for (auto i { 0u }; i < fn.values.size(); ++i) m_value_regs[i] = alloc_vreg(fn.values[i].type);
36
37 visit_function(fn);
38
39 m_fn = nullptr;
40 return std::move(m_builder).finalise();
41 }

57 void on_inst(ir::IRFunction& fn, ir::IRBlock&, ir::IRInst& inst) {
58 using enum ir::IROp;
59 switch (inst.op) {
60 case iconst: lower_iconst(inst); break;
61 case iadd: lower_binop(inst, AluOp::Add); break;
62 case isub: lower_binop(inst, AluOp::Sub); break;
63 case imul: lower_binop(inst, AluOp::Mul); break;
64 case ret: lower_ret(inst); break;
65 case idiv:
66 case icmp:
67 case br:
68 case brif:
69 case call: std::unreachable();
70 }
71 }

94 void lower_iconst(const ir::IRInst& inst) {
95 m_builder.push(inst::MovImm{

101 void lower_binop(const ir::IRInst& inst, AluOp op) {
102 const auto dst = reg_for(inst.result);
103 const auto lhs = reg_for(m_fn->operands[inst.ops_offset]);
104 const auto rhs = reg_for(m_fn->operands[inst.ops_offset + 1]);
105
106 m_builder.push(inst::Mov{ .dst = dst, .src = lhs });
107 m_builder.push(inst::Alu{ .op = op, .dst = dst, .src = rhs });
108 }
109
110 void lower_ret(const ir::IRInst& inst) {
111 if (inst.ops_count > 0) {
112 const auto v = reg_for(m_fn->operands[inst.ops_offset]);
113 m_builder.push(inst::Mov{
114 .dst = codegen::Reg::from_preg(RAX),
115 .src = v,
116 });
117 }
118
119 m_builder.push(inst::Ret {});
120 }
