36 [[nodiscard]] static constexpr MachOperand any(
37 const VReg vreg,
38 const OperandKind kind,
39 const OperandPos pos
40 ) noexcept {
41 MachOperand op;
42 op.bits = (static_cast<uint64_t>(vreg.index()) << vreg_index_shift)
43 | (static_cast<uint64_t>(vreg.rc()) << vreg_class_shift)
44 | (static_cast<uint64_t>(kind) << kind_shift)
45 | (static_cast<uint64_t>(pos) << pos_shift);
46 return op;
47 }

50 [[nodiscard]] static constexpr MachOperand fixed(
51 const VReg vreg,
52 const OperandKind kind,
53 const OperandPos pos,
54 const PReg preg
55 ) noexcept {
56 assert(preg.valid());
57 MachOperand op = any(vreg, kind, pos);
58 op.bits |= (1ull << fixed_flag_shift)
59 | (static_cast<uint64_t>(preg.bits) << preg_shift);
60 return op;
61 }

63 [[nodiscard]] constexpr VReg vreg() const noexcept {
64 VReg r;
65 r.bits = static_cast<uint32_t>(
66 ((bits & vreg_index_mask) >> vreg_index_shift) << 1 |
67 ((bits & vreg_class_mask) >> vreg_class_shift)
68 );
69 return r;
70 }
71
72 [[nodiscard]] constexpr OperandKind kind() const noexcept {
73 return static_cast<OperandKind>((bits & kind_mask) >> kind_shift);
74 }
75
76 [[nodiscard]] constexpr OperandPos pos() const noexcept {
77 return static_cast<OperandPos>((bits & pos_mask) >> pos_shift);
78 }
79
80 [[nodiscard]] constexpr bool is_fixed() const noexcept {
81 return (bits & fixed_flag_mask) != 0;
82 }
83
85 [[nodiscard]] constexpr std::optional<PReg> fixed_preg() const noexcept {
86 if (!is_fixed()) return std::nullopt;
87 PReg r;
88 r.bits = static_cast<uint8_t>((bits & preg_mask) >> preg_shift);
89 return r;
90 }
91
92 [[nodiscard]] constexpr bool operator==(const MachOperand &) const noexcept = default;
93 private:
94 static constexpr int vreg_index_shift = 43;
95 static constexpr int vreg_class_shift = 42;
96 static constexpr int kind_shift = 40;
97 static constexpr int pos_shift = 39;
98 static constexpr int fixed_flag_shift = 38;
99 static constexpr int preg_shift = 0;
100 static constexpr uint64_t vreg_index_mask = ((1ull << 21) - 1) << vreg_index_shift;
101 static constexpr uint64_t vreg_class_mask = 1ull << vreg_class_shift;
102 static constexpr uint64_t kind_mask = 0b11ull << kind_shift;
103 static constexpr uint64_t pos_mask = 1ull << pos_shift;
104 static constexpr uint64_t fixed_flag_mask = 1ull << fixed_flag_shift;
105 static constexpr uint64_t preg_mask = 0xFFull << preg_shift;
106 };
